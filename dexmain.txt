function timeAgo(dateStr: string) {
  const diff = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000);
  if (diff < 60) return `${diff}s`;
  if (diff < 3600) return `${Math.floor(diff / 60)}m`;
  if (diff < 86400) return `${Math.floor(diff / 3600)}h`;
  return `${Math.floor(diff / 86400)}d`;
}


export default function DexPageContent() {
  const searchParams = useSearchParams();
  const mintFromUrl = searchParams.get("mint");
  const [searchInput, setSearchInput] = useState("");

  const {
    search,
    selectedMint,
    selectedGenesisAccount,
    isBuy,
    amount,
    setSearch,
    selectToken,
    setIsBuy,
    setAmount,
    resetTrade,
  } = useDexStore();

  const { tokens, total, isLoading, isFetching, isError, error, refresh } = useDexTokens();
  const { swap, isSwapping, error: swapError, setError: setSwapError } = useSwap();
  const [mobileTradeOpen, setMobileTradeOpen] = useState(false);

  const { trades, isLoading: isLoadingTrades } = useTrades(selectedMint);

  const filteredTokens = useMemo(() => {
    const base = filterTokens(tokens, search);
    return [...base].sort((a, b) => (b.createdAt ?? 0) - (a.createdAt ?? 0));
  }, [tokens, search]);

  const selectedToken = useMemo(
    () => tokens.find((t) => t.mint === selectedMint) ?? null,
    [tokens, selectedMint]
  );

  const genesisAccountForCurve =
    selectedGenesisAccount ?? selectedToken?.genesisAccount ?? selectedToken?.mint ?? null;

  const { data: curveInfo, isLoading: isLoadingCurve } = useBondingCurveInfo(genesisAccountForCurve);

  useEffect(() => {
    const match = tokens.find((t) => t.mint === mintFromUrl);
    if (match) selectToken(match.mint, match.genesisAccount ?? null);
  }, [mintFromUrl, tokens, selectToken]);

  const handleSelectToken = (token: typeof tokens[0]) => {
    selectToken(token.mint, token.genesisAccount ?? null);
    setSwapError("");
    if (window.innerWidth < 1024) setMobileTradeOpen(true);
  };

  const handleSwap = async () => {
    if (!selectedToken || !amount) return;
    const ok = await swap({
      genesisAccount: selectedGenesisAccount ?? selectedToken.genesisAccount ?? selectedToken.mint,
      mint: selectedToken.mint,
      amount,
      isBuy,
    });
    if (ok) { resetTrade(); setMobileTradeOpen(false); }
  };

  const fillPercent = curveInfo?.lifecycle?.fillPercent ?? 0;
  const priceRaw = curveInfo?.price?.tokensPerSol ? Number(curveInfo.price.tokensPerSol) : null;
  const priceText = priceRaw && Number.isFinite(priceRaw)
    ? priceRaw.toLocaleString(undefined, { maximumFractionDigits: 4 })
    : "â€”";

    token: selectedToken,
    isBuy, amount, isSwapping, swapError,
    curveInfo: curveInfo ?? null,
    isLoadingCurve,
    onToggleBuy: setIsBuy,
    onAmountChange: setAmount,
    onSwap: handleSwap,
  };

      {/* AMBIENT BG */}
      <div style={{ position: "fixed", inset: 0, pointerEvents: "none", zIndex: 0, overflow: "hidden" }}>
        <div style={{
          position: "absolute", top: "-10%", left: "-5%",
          width: 600, height: 600, borderRadius: "50%",
          background: "radial-gradient(circle, rgba(153,69,255,0.07) 0%, transparent 70%)",
        }} />
        <div style={{
          position: "absolute", top: "40%", right: "-10%",
          width: 500, height: 500, borderRadius: "50%",
          background: "radial-gradient(circle, rgba(20,241,149,0.04) 0%, transparent 70%)",
        }} />
      </div>

      <div style={{ position: "relative", zIndex: 1 }}>
        <DexHeader onRefresh={() => refresh()} isRefreshing={isFetching} />

        {/* MAIN 3-COLUMN LAYOUT */}
        <div style={{
          display: "grid",
          gridTemplateColumns: "280px 1fr 320px",
          gridTemplateRows: "auto 1fr",
          gap: 0,
          height: "calc(100vh - 64px)",
          overflow: "hidden",
        }} className="dex-grid">

          {/* LEFT: TOKEN LIST */}
          <div style={{
            borderRight: "1px solid rgba(153,69,255,0.1)",
            display: "flex",
            flexDirection: "column",
            overflow: "hidden",
          }}>
            <div style={{ padding: "12px 12px 8px", borderBottom: "1px solid rgba(153,69,255,0.08)" }}>
              <div style={{ position: "relative" }}>
                <span style={{
                  position: "absolute", left: 10, top: "50%", transform: "translateY(-50%)",
                  color: "rgba(153,69,255,0.4)", fontSize: 12, pointerEvents: "none",
                }}>â¬¡</span>
                <input
                  value={searchInput}
                  onChange={e => { setSearchInput(e.target.value); setSearch(e.target.value); }}
                  placeholder="Search tokens..."
                  style={{
                    width: "100%", height: 36,
                    paddingLeft: 28, paddingRight: 12,
                    background: "rgba(153,69,255,0.06)",
                    border: "1px solid rgba(153,69,255,0.15)",
                    borderRadius: 8,
                    color: "#fff", fontSize: 12,
                    fontFamily: "monospace",
                    outline: "none",
                    boxSizing: "border-box",
                  }}
                />
              </div>
            </div>

            <div style={{
              display: "grid", gridTemplateColumns: "42px 1fr 50px",
              gap: 8, padding: "8px 12px",
              borderBottom: "1px solid rgba(153,69,255,0.08)",
            }}>
              {["", "TOKEN", "AGE"].map(h => (
                <span key={h} style={{ color: "rgba(153,69,255,0.35)", fontFamily: "monospace", fontSize: 9, letterSpacing: "0.12em" }}>{h}</span>
              ))}
            </div>

            <div style={{ flex: 1, overflowY: "auto", padding: "4px 8px" }} className="custom-scroll">
              {isLoading ? (
                <div style={{ padding: 20, textAlign: "center" }}>
                  <span style={{ color: "rgba(153,69,255,0.4)", fontFamily: "monospace", fontSize: 11 }}>Loading...</span>
                </div>
              ) : filteredTokens.length === 0 ? (
                <div style={{ padding: 20, textAlign: "center" }}>
                  <span style={{ color: "rgba(153,69,255,0.2)", fontFamily: "monospace", fontSize: 11 }}>NO_TOKENS</span>
                </div>
              ) : (
                filteredTokens.map(token => (
                  <div
                    key={token.mint}
                    onClick={() => handleSelectToken(token)}

                      cursor: "pointer",
                      borderRadius: 8,
                      background: selectedMint === token.mint ? "rgba(153,69,255,0.1)" : "transparent",
                      border: selectedMint === token.mint ? "1px solid rgba(153,69,255,0.2)" : "1px solid transparent",

                  </div>
                ))
              )}
            </div>
          </div>

          {/* CENTER: CHART + TRADES */}

            borderRight: "1px solid rgba(153,69,255,0.1)",
            display: "flex",
            flexDirection: "column",
            overflow: "hidden",
          }}>
            {selectedToken ? (
              <>
                {/* Token header */}
                <div style={{
                  padding: "12px 20px",
                  borderBottom: "1px solid rgba(153,69,255,0.1)",
                  display: "flex", alignItems: "center", gap: 16, flexWrap: "wrap",
                  background: "rgba(153,69,255,0.03)",
                }}>
                  <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
                    {selectedToken.imageUrl ? (
                      <img src={selectedToken.imageUrl} alt={selectedToken.symbol}
                        style={{ width: 36, height: 36, borderRadius: "50%", objectFit: "cover", border: "1.5px solid rgba(153,69,255,0.3)" }} />
                    ) : (
                      <div style={{
                        width: 36, height: 36, borderRadius: "50%",
                        background: "linear-gradient(135deg, #9945FF, #14F195)",
                        display: "flex", alignItems: "center", justifyContent: "center",
                        fontWeight: 900, color: "#fff", fontSize: 14,
                      }}>{selectedToken.symbol[0]}</div>
                    )}
                    <div>
                      <div style={{ display: "flex", alignItems: "center", gap: 6 }}>

                        <span style={{ color: "rgba(153,69,255,0.8)", fontFamily: "monospace", fontSize: 11, background: "rgba(153,69,255,0.1)", padding: "1px 6px", borderRadius: 4 }}>${selectedToken.symbol}</span>
                      </div>
                      <span style={{ color: "rgba(255,255,255,0.2)", fontFamily: "monospace", fontSize: 10 }}>
                        {selectedToken.mint.slice(0, 6)}...{selectedToken.mint.slice(-4)}
                      </span>
                    </div>
                  </div>

                  <div style={{ display: "flex", gap: 20, marginLeft: 8, flexWrap: "wrap" }}>

                      { label: "PRICE", value: priceText ? `${priceText} ${selectedToken.symbol}/SOL` : "â€”" },
                      { label: "CURVE", value: isLoadingCurve ? "Â·Â·Â·" : `${fillPercent.toFixed(2)}%` },
                      { label: "TRADES", value: trades.length.toString() },
                    ].map(s => (

                        <div style={{ color: "rgba(255,255,255,0.25)", fontFamily: "monospace", fontSize: 9, letterSpacing: "0.1em" }}>{s.label}</div>
                        <div style={{ color: s.label === "CURVE" && fillPercent > 80 ? "#14F195" : "rgba(255,255,255,0.8)", fontFamily: "monospace", fontSize: 12, fontWeight: 700 }}>{s.value}</div>
                      </div>
                    ))}
                  </div>

                  <div style={{ marginLeft: "auto", display: "flex", alignItems: "center", gap: 6 }}>
                    <div style={{ width: 7, height: 7, borderRadius: "50%", background: "#14F195", boxShadow: "0 0 8px #14F195", animation: "pulse 2s infinite" }} />
                    <span style={{ color: "rgba(255,255,255,0.3)", fontFamily: "monospace", fontSize: 10 }}>LIVE</span>
                  </div>
                </div>

                {/* CHART */}
                <div style={{ flex: 1, overflow: "hidden", position: "relative", minHeight: 200 }}>
                  <div style={{
                    position: "absolute", inset: 0,
                    backgroundImage: "linear-gradient(rgba(153,69,255,0.025) 1px, transparent 1px), linear-gradient(90deg, rgba(153,69,255,0.025) 1px, transparent 1px)",
                    backgroundSize: "48px 48px",
                  }} />
                  <div style={{ position: "absolute", inset: 0, padding: 8 }}>
                    <TradeChart symbol={selectedToken?.symbol ?? ""} mint={selectedToken?.mint ?? ""} />
                  </div>
                </div>

                {/* RECENT TRADES */}
                <div style={{
                  borderTop: "1px solid rgba(153,69,255,0.1)",
                  height: 220, overflow: "hidden", display: "flex", flexDirection: "column",
                }}>
                  <div style={{
                    padding: "8px 16px",
                    borderBottom: "1px solid rgba(153,69,255,0.08)",
                    display: "flex", alignItems: "center", gap: 8,
                    background: "rgba(153,69,255,0.03)",
                  }}>
                    <div style={{ width: 6, height: 6, borderRadius: "50%", background: "#14F195", boxShadow: "0 0 6px #14F195", animation: "pulse 1.5s infinite" }} />
                    <span style={{ color: "rgba(153,69,255,0.6)", fontFamily: "monospace", fontSize: 9, letterSpacing: "0.14em" }}>RECENT TRADES</span>

                  </div>

                  {/* Table header */}
                  <div style={{
                    display: "grid", gridTemplateColumns: "55px 90px 1fr 60px 110px",
                    padding: "6px 16px",
                    borderBottom: "1px solid rgba(153,69,255,0.06)",
                  }}>
                    {["TYPE", "SOL", "TOKENS", "TIME", "MAKER"].map(h => (
                      <span key={h} style={{ color: "rgba(153,69,255,0.3)", fontFamily: "monospace", fontSize: 9, letterSpacing: "0.1em" }}>{h}</span>
                    ))}
                  </div>

                  <div style={{ flex: 1, overflowY: "auto" }} className="custom-scroll">
                    {isLoadingTrades ? (
                      <div style={{ display: "flex", alignItems: "center", justifyContent: "center", height: "100%" }}>
                        <span style={{ color: "rgba(153,69,255,0.3)", fontFamily: "monospace", fontSize: 10 }}>Loading...</span>
                      </div>
                    ) : trades.length === 0 ? (
                      <div style={{ display: "flex", alignItems: "center", justifyContent: "center", height: "100%" }}>
                        <span style={{ color: "rgba(255,255,255,0.08)", fontFamily: "monospace", fontSize: 10, letterSpacing: "0.08em" }}>
                          AWAITING_TRANSACTIONS...
                        </span>
                      </div>
                    ) : (
                      trades.map(trade => (
                        <div key={trade.id} style={{
                          display: "grid", gridTemplateColumns: "55px 90px 1fr 60px 110px",
                          padding: "6px 16px",
                          borderBottom: "1px solid rgba(153,69,255,0.04)",
                          alignItems: "center",
                          transition: "background 0.1s",
                        }}
                        onMouseEnter={e => (e.currentTarget.style.background = "rgba(153,69,255,0.04)")}
                        onMouseLeave={e => (e.currentTarget.style.background = "transparent")}
                        >
                          <span style={{
                            fontFamily: "monospace", fontSize: 10, fontWeight: 700,
                            color: trade.is_buy ? "#14F195" : "#ff2d95",
                            letterSpacing: "0.06em",
                          }}>
                            {trade.is_buy ? "BUY" : "SELL"}
                          </span>
                          <span style={{ fontFamily: "monospace", fontSize: 10, color: "rgba(255,255,255,0.7)" }}>
                            {trade.amount_sol.toFixed(4)} SOL
                          </span>
                          <span style={{ fontFamily: "monospace", fontSize: 10, color: "rgba(255,255,255,0.5)" }}>
                            {trade.amount_token > 1000
                              ? `${(trade.amount_token / 1000).toFixed(1)}K`
                              : trade.amount_token.toFixed(2)}
                          </span>
                          <span style={{ fontFamily: "monospace", fontSize: 10, color: "rgba(255,255,255,0.3)" }}>
                            {timeAgo(trade.created_at)}
                          </span>
                          <a
                            href={trade.tx_signature ? `https://solscan.io/tx/${trade.tx_signature}` : "#"}
                            target="_blank"
                            rel="noopener noreferrer"
                            style={{
                              fontFamily: "monospace", fontSize: 10,
                              color: "rgba(153,69,255,0.6)",
                              textDecoration: "none",
                              overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap",
                            }}
                          >
                            {trade.wallet.slice(0, 4)}...{trade.wallet.slice(-4)}
                          </a>
                        </div>
                      ))
                    )}
                  </div>
                </div>
              </>
            ) : (
              <div style={{
                flex: 1, display: "flex", flexDirection: "column",
                alignItems: "center", justifyContent: "center", gap: 16, padding: 40,
              }}>
                <div style={{ fontSize: 48, filter: "drop-shadow(0 0 20px rgba(153,69,255,0.3))", opacity: 0.3 }}>â—ˆ</div>
                <p style={{ color: "rgba(153,69,255,0.3)", fontFamily: "monospace", fontSize: 12, letterSpacing: "0.1em", textAlign: "center" }}>
                  SELECT A TOKEN<br />
                  <span style={{ color: "rgba(255,255,255,0.1)", fontSize: 10 }}>chart and activity will appear here</span>
                </p>
              </div>
            )}
          </div>

          {/* RIGHT: TRADE PANEL */}
          <div style={{ overflowY: "auto", padding: 16 }} className="custom-scroll">
            <TradePanel {...tradePanelProps} />
          </div>
        </div>

        {/* Mobile bottom bar */}
        {selectedToken && (
          <div style={{
            display: "none",
            position: "fixed", bottom: 0, left: 0, right: 0, zIndex: 40,
            padding: "12px 16px",
            background: "rgba(7,7,15,0.97)",
            borderTop: "1px solid rgba(153,69,255,0.15)",

            <button
              onClick={() => setMobileTradeOpen(true)}
              style={{
                width: "100%", height: 50, borderRadius: 14,
                background: isBuy ? "linear-gradient(135deg, #14F195, #0fa96a)" : "linear-gradient(135deg, #ff2d95, #c4006b)",
                border: "none", color: "#fff",
                fontWeight: 900, fontSize: 14, letterSpacing: "0.08em",
                fontFamily: "monospace", cursor: "pointer",
              }}>
              TRADE {selectedToken.symbol}
            </button>
          </div>
        )}

        <Footer />
      </div>

      <TokenModal
        token={selectedToken}
        open={mobileTradeOpen}
        curveInfo={curveInfo ?? null}
        isLoadingCurve={isLoadingCurve}
        onClose={() => setMobileTradeOpen(false)}
      >
        <TradePanel {...tradePanelProps} showDetail={false} compact />
      </TokenModal>

      <style>{`
        @keyframes pulse { 0%,100%{opacity:1} 50%{opacity:0.3} }
        .custom-scroll::-webkit-scrollbar { width: 4px; }
        .custom-scroll::-webkit-scrollbar-track { background: transparent; }
        .custom-scroll::-webkit-scrollbar-thumb { background: rgba(153,69,255,0.2); border-radius: 2px; }
        .custom-scroll::-webkit-scrollbar-thumb:hover { background: rgba(153,69,255,0.4); }
        @media (max-width: 1024px) {
          .dex-grid { grid-template-columns: 1fr !important; height: auto !important; overflow: visible !important; }
          .mobile-trade-bar { display: block !important; }
        }
      `}</style>
    </div>
  );
