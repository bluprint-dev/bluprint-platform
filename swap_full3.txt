import { NextRequest, NextResponse } from "next/server";
import { publicKey, transactionBuilder, createNoopSigner, signerIdentity } from "@metaplex-foundation/umi";
import { findAssociatedTokenPda, createAssociatedToken } from "@metaplex-foundation/mpl-toolbox";
import { getPlatformUmi } from "@/app/lib/umi";

import {
  findBondingCurveBucketV2Pda,
  fetchBondingCurveBucketV2,
  getSwapResult,
  swapBondingCurveV2,
  SwapDirection,
  isSwappable,
} from "@metaplex-foundation/genesis";

const WSOL_MINT = "So11111111111111111111111111111111111111112";

const PLATFORM_FEE_WALLET = process.env.BONDING_CURVE_FEE_WALLET ?? "";

function toBigIntSafe(value: unknown): bigint {
  try {
    if (typeof value === "bigint") return value;
    return BigInt(value as string | number);
  } catch {
    throw new Error("INVALID_BIGINT");
  }
}

function safeError(error: unknown) {
  if (error instanceof Error) {
    return { name: error.name, message: error.message };
  }
  return { message: String(error) };
}

function minOutWithSlippage(amountOut: bigint) {
  return (amountOut * BigInt(99)) / BigInt(100);
}

function extractBaseMint(bucket: unknown): string | null {
  try {
    const b = (bucket as any).bucket;
    const baseMint = b?.baseMint;
    if (!baseMint) return null;
    if (baseMint.__option === "Some" && baseMint.value) return baseMint.value;
    if (typeof baseMint === "string") return baseMint;
    return null;
  } catch {
    return null;
  }
}

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    console.log("SWAP_REQUEST_BODY:", body);

    const {
      genesisAccount: genesisAccountRaw,
      mintAddress,
      amount,
      userPublicKey,
      isBuy,
    } = body;

    if (!genesisAccountRaw || !userPublicKey || !amount) {
      return NextResponse.json(
        { success: false, error: "MISSING_PARAMS â€” genesisAccount, userPublicKey, amount required" },
        { status: 400 }
      );
    }

    const amountBigInt = toBigIntSafe(amount);
    const umi = getPlatformUmi().use(signerIdentity(createNoopSigner(user)));

    const genesisAccount = publicKey(genesisAccountRaw);
    const user = publicKey(userPublicKey);
    const wsol = publicKey(WSOL_MINT);

    // â€”â€” BUCKET â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”
    const [bucketPda] = findBondingCurveBucketV2Pda(umi, {
      genesisAccount,
      bucketIndex: 0,
    });

    let bucket;
    try {
      bucket = await fetchBondingCurveBucketV2(umi, bucketPda);
    } catch (e) {
      return NextResponse.json(
        { success: false, error: "BUCKET_FETCH_FAILED â€” genesisAccount may be invalid" },
        { status: 400 }
      );
    }

    if (!bucket || !isSwappable(bucket)) {
      return NextResponse.json(
        { success: false, error: "NOT_SWAPPABLE" },
        { status: 400 }
      );
    }

    const direction = isBuy ? SwapDirection.Buy : SwapDirection.Sell;

    // â€”â€” QUOTE â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”
    const quote = getSwapResult(bucket, amountBigInt, direction);
    const minOut = minOutWithSlippage(quote.amountOut);

    // â€”â€” BASE MINT â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”
    const baseMintStr = extractBaseMint(bucket) ?? mintAddress ?? null;
    if (!baseMintStr) {
      return NextResponse.json(
        { success: false, error: "MISSING_BASE_MINT â€” provide mintAddress in request body" },
        { status: 400 }
      );
    }
    const baseMint = publicKey(baseMintStr);

    // â€”â€” USER ATAs â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”
    const [baseATA] = findAssociatedTokenPda(umi, { mint: baseMint, owner: user });
    const [quoteATA] = findAssociatedTokenPda(umi, { mint: wsol, owner: user });

    // â€”â€” PLATFORM FEE ATA â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”
    // ATA'yÄ± ayrÄ± tx ile oluÅŸturmak yerine swap tx'ine instruction olarak ekliyoruz.
    // BÃ¶ylece platform wallet bakiyesi sÄ±fÄ±r olsa bile sorun olmaz â€”
    // ATA oluÅŸturma maliyeti swap yapan kullanÄ±cÄ± tarafÄ±ndan karÅŸÄ±lanÄ±r (sadece ilk swap'ta, ~0.002 SOL).
    // Sonraki swap'larda ATA zaten var olacaÄŸÄ± iÃ§in instruction idempotent olarak Ã§alÄ±ÅŸÄ±r.
    let feeQuoteTokenAccount: ReturnType<typeof publicKey> | undefined = undefined;
    let feeAtaBuilder = transactionBuilder();

    if (PLATFORM_FEE_WALLET) {
      const platformWallet = publicKey(PLATFORM_FEE_WALLET);
      const [feeAta] = findAssociatedTokenPda(umi, {
        mint: wsol,
        owner: platformWallet,
      });
      feeQuoteTokenAccount = feeAta;

      // Onchain'de var mÄ± kontrol et
      const feeAtaAccount = await umi.rpc.getAccount(feeAta);
      if (!feeAtaAccount.exists) {
        // Yoksa swap tx'ine createAssociatedToken instruction ekle (idempotent)
        console.log("FEE_ATA not found, will create inside swap tx:", feeAta.toString());
        feeAtaBuilder = createAssociatedToken(umi, {
  mint: wsol,
  owner: platformWallet,
  payer: user,
});
      } else {
        console.log("FEE_ATA exists:", feeAta.toString());
      }
    } else {
      console.warn("PLATFORM_FEE_WALLET not set â€” creator fee will not be collected");
    }

    // â€”â€” BUILD SWAP â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”â€”
    const swapBuilder = swapBondingCurveV2(umi, {
      genesisAccount,
      bucket: bucketPda,
      baseMint,
      quoteMint: wsol,
      baseTokenAccount: baseATA,
      quoteTokenAccount: quoteATA,
      baseTokenOwner: user,
      quoteTokenOwner: user,
      swapDirection: direction,
      amount: amountBigInt,
      minAmountOutScaled: minOut,
      ...(feeQuoteTokenAccount ? { feeQuoteTokenAccount } : {}),
    });

    // feeAtaBuilder varsa Ã¶nce ekle, sonra swap â€” tek tx
    const combinedBuilder = feeAtaBuilder.add(swapBuilder);

    const tx = await (await combinedBuilder.setLatestBlockhash(umi)).buildAndSign(umi);
    const serialized = Buffer.from(umi.transactions.serialize(tx)).toString("base64");

    return NextResponse.json({
      success: true,
      transaction: serialized,
      quote: {
        amountIn: quote.amountIn.toString(),
        amountOut: quote.amountOut.toString(),
        fee: quote.fee.toString(),
        creatorFee: quote.creatorFee?.toString() ?? "0",
      },
      meta: {
        genesisAccount: genesisAccountRaw,
        mintAddress: baseMintStr,
        userPublicKey,
        direction: isBuy ? "buy" : "sell",
        feeWallet: PLATFORM_FEE_WALLET || null,
      },
    });
  } catch (error) {
    console.error("SWAP_FATAL:", safeError(error));
    return NextResponse.json(
      { success: false, error: "SWAP_FAILED", details: safeError(error) },
      { status: 500 }
    );
  }
}
