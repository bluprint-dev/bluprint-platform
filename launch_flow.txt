// -----------------------------
// UMI INIT

function getPlatformUmi() {
  const rpcUrl = process.env.NEXT_PUBLIC_RPC_URL;
  if (!rpcUrl) throw new Error("NEXT_PUBLIC_RPC_URL missing");

  const secretKeyRaw = process.env.PLATFORM_SECRET_KEY;
  if (!secretKeyRaw) throw new Error("PLATFORM_SECRET_KEY missing");

  const umi = createUmi(rpcUrl).use(genesis());

  const secretKey = Uint8Array.from(JSON.parse(secretKeyRaw));
  const keypair = umi.eddsa.createKeypairFromSecretKey(secretKey);

  umi.use(keypairIdentity(keypair));
  return umi;
}

// -----------------------------
function safeError(error: unknown) {
  if (error instanceof Error) {
    return {
      name: error.name,
      message: error.message,
      stack: error.stack,
    };
  }
  return { message: String(error) };
}

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();

    console.log("LAUNCH_REQUEST:", body);

    const {
      name,
      symbol,
      image,
      description,
      twitter,
      telegram,
      website,
    } = body;

    if (!name || !symbol || !image) {
      return NextResponse.json(
        { success: false, error: "Missing required fields" },
        { status: 400 }
      );
    }

    const umi = getPlatformUmi();

    const input = {
      wallet: umi.identity.publicKey,

      token: {
        name,
        symbol,
        image,
        description: description || "",
        externalLinks: {
          website: website || undefined,
          twitter: twitter || undefined,
          telegram: telegram || undefined,
        },
      },

      launchType: "bondingCurve" as const,

      launch: {
        creatorFeeWallet: umi.identity.publicKey,
        firstBuyAmount: 0,
      },
    };

    console.log("CREATE_INPUT:", input);

    const result = await createAndRegisterLaunch(umi, {}, input);

    console.log("CREATE_RESULT:", result);

    // ======================================================
    // âœ… FIX: genesisAccount kaydediliyor (mint deÄŸil)
    // bonding-curve:tokens set'i curve operasyonlarÄ± iÃ§in kullanÄ±lÄ±r.
    // PDA tÃ¼retme, bucket fetch, swap, quote hepsi genesisAccount ister.
    // ======================================================

    await redis.sadd(
      "bonding-curve:tokens",
      result.genesisAccount  // âœ… genesisAccount â€” mint deÄŸil
    );

    // Legacy set'e de genesisAccount yaz
    await redis.sadd("bonding-curve:tokens:legacy", result.genesisAccount);

    // genesisAccount â†’ mintAddress lookup (reverse map)
    await redis.set(
      `genesis:mint:${result.genesisAccount}`,
      result.mintAddress
    );

    console.log("LAUNCH_ADDRESSES", {
      mintAddress: result.mintAddress,
      genesisAccount: result.genesisAccount,
    });

    // ======================================================
    // METADATA: mint bazlÄ± anahtar â€” DOÄRU
    // token:metadata:${mint} display/metadata iÃ§in mint ile kalÄ±r
    // genesisAccount da metadata iÃ§ine eklendi
    // ======================================================
    await redis.set(
      `token:metadata:${result.mintAddress}`,
      JSON.stringify({
        mint: result.mintAddress,
        genesisAccount: result.genesisAccount,  // âœ… her zaman sakla

        name,
        symbol,
        imageUrl: image,

        description: description || "",

        creator: umi.identity.publicKey.toString(),

        website: website || "",
        twitter: twitter || "",
        telegram: telegram || "",

        createdAt: Date.now(),
      })
    );

    // genesisAccount â†’ metadata lookup iÃ§in ek index
    await redis.set(
      `genesis:metadata:${result.genesisAccount}`,
      result.mintAddress  // mint'e pointer â€” metadata fetch iÃ§in
    );

    console.log("REDIS_SAVED:", {
      metadataKey: `token:metadata:${result.mintAddress}`,
      genesisKey: `genesis:mint:${result.genesisAccount}`,
    });

    return NextResponse.json({
      success: true,

      mintAddress: result.mintAddress,
      genesisAccount: result.genesisAccount,

      launchId: result.launch.id,
      launchLink: result.launch.link,

      token: {
        mint: result.mintAddress,
        genesisAccount: result.genesisAccount,
      },
    });
  } catch (err) {
    console.error("LAUNCH_FATAL:", safeError(err));

    if (isGenesisValidationError(err)) {
      return NextResponse.json(
        {
          success: false,
          type: "VALIDATION_ERROR",
          field: err.field,
          error: err.message,
        },
        { status: 400 }
      );
    }

    if (isGenesisApiError(err)) {
      return NextResponse.json(
        {
          success: false,
          type: "GENESIS_API_ERROR",
          statusCode: err.statusCode,
          details: err.responseBody,
        },
        { status: 502 }
      );
    }

    if (isGenesisApiNetworkError(err)) {
      return NextResponse.json(
        {
          success: false,
          type: "NETWORK_ERROR",
          error: err.cause.message,
        },
        { status: 503 }
      );
    }

    return NextResponse.json(
      {
        success: false,
        type: "UNKNOWN_ERROR",
        error: err instanceof Error ? err.message : "Unknown error",
      },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from 'next/server';
import { redis, KEYS } from '@/app/lib/redis';

const REFERRAL_REWARD = 0.05; // SOL

export async function POST(req: NextRequest) {
  try {
    const {
      mintAddress,
      genesisAccount,   // âœ… ayrÄ± alan â€” bonding curve operasyonlarÄ± iÃ§in
      name,
      symbol,
      imageUrl,
      userPublicKey,
      signature,
    } = await req.json();

    // âœ… FIX: genesisAccount da zorunlu
    if (!mintAddress || !genesisAccount || !userPublicKey || !signature) {
      return NextResponse.json({ error: 'Missing fields' }, { status: 400 });
    }

    // ===============================
    // RATE LIMIT
    // ===============================
    const rateKey = `rate-limit:${userPublicKey}`;
    const rate = await redis.incr(rateKey);

    if (rate === 1) await redis.expire(rateKey, 300);
    if (rate > 3) {
      return NextResponse.json({ error: 'Rate limit exceeded. Wait 5 minutes.' }, { status: 429 });
    }

    // ===============================
    // DUPLICATE PROTECTION
    // âœ… FIX: genesisAccount bazlÄ± key â€” curve operasyonlarÄ±nda bu key kullanÄ±lÄ±r
    // ===============================
    const existing = await redis.get(`bonding-curve:creator:${genesisAccount}`);
    if (existing) {
      return NextResponse.json({ error: 'Token already exists' }, { status: 400 });
    }

    // ===============================
    // FIRST TOKEN CHECK (ANTI FARM)
    // ===============================
    const userTokenCountKey = `user:token-count:${userPublicKey}`;
    const tokenCount = await redis.incr(userTokenCountKey);
    if (tokenCount === 1) await redis.expire(userTokenCountKey, 86400 * 30);

    // ===============================
    // SAVE TOKEN
    // âœ… FIX: tÃ¼m curve key'leri genesisAccount ile
    //         metadata key'leri mint ile (display iÃ§in)
    // ===============================

    // curve creator kaydÄ± â†’ genesisAccount bazlÄ±
    await redis.set(`bonding-curve:creator:${genesisAccount}`, userPublicKey);

    // âœ… FIX: registry set'ine genesisAccount yazÄ±lÄ±yor (mint deÄŸil)
    // tokenRegistry.ts bu set'i okuyup her deÄŸeri genesisAccount olarak kullanÄ±r
    await redis.sadd('bonding-curve:tokens', genesisAccount);

    // genesisAccount â†’ mintAddress reverse lookup
    await redis.set(`genesis:mint:${genesisAccount}`, mintAddress);

    // metadata: mint bazlÄ± anahtar DOÄRU â€” display/image/symbol iÃ§in mint kullanÄ±lÄ±r
    await redis.set(
      `token:metadata:${mintAddress}`,
      JSON.stringify({
        mint: mintAddress,
        genesisAccount,        // âœ… metadata iÃ§inde sakla
        name,
        symbol,
        imageUrl,
        creator: userPublicKey,
        createdAt: Date.now(),
        signature,
      })
    );

    // ===============================
    // TRENDING
    // âœ… FIX: trending'e de genesisAccount yaz
    // ===============================
    await redis.zincrby('trending:tokens', 1, genesisAccount);
    await redis.expire('trending:tokens', 86400);

    // ===============================
    // REFERRAL ENGINE
    // ===============================
    const referralKey = `referral:by:${userPublicKey}`;
    const referrer = await redis.get(referralKey);

    if (referrer && tokenCount === 1) {
      // only first token reward
      const earningsKey = `${KEYS.earnings}:${referrer}`;
      const earningsRaw = await redis.get(earningsKey);

      const earnings = earningsRaw
        ? JSON.parse(earningsRaw as string)
        : { pending: 0, claimed: 0, referrals: [] };

      // prevent duplicate referral reward per user
      if (!earnings.referrals.includes(userPublicKey)) {
        earnings.pending += REFERRAL_REWARD;
        earnings.referrals.push(userPublicKey);

        await redis.set(earningsKey, JSON.stringify(earnings));

        // optional leaderboard boost
        await redis.zincrby(KEYS.boostLeaderboard, 1, referrer);
      }
    }

    console.log('âœ… Token tracked + referral processed:', {
      mintAddress,
      genesisAccount,
    });

    return NextResponse.json({ success: true, mintAddress, genesisAccount });

  } catch (error: any) {
    console.error('Track error:', error);
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

// ==================== SLIDING WINDOW RATE LIMITER ====================
export interface RateLimitResult {
  success: boolean;
  limit: number;
  remaining: number;
  retryAfter?: number;
  resetTime?: number;
}

export async function rateLimitByIP(
  ip: string, 
  limit: number = 20, 
  windowMs: number = 60 * 1000
): Promise<RateLimitResult> {
  const now = Date.now();
  const windowStart = now - windowMs;
  const key = `ratelimit:${ip}`;
  
  try {
    // 1. Window dÄ±ÅŸÄ±ndaki eski kayÄ±tlarÄ± temizle
    await redis.zremrangebyscore(key, 0, windowStart);
    
    // 2. Mevcut istek sayÄ±sÄ±nÄ± al
    const count = await redis.zcard(key);
    
    // 3. Limit kontrolÃ¼
    if (count >= limit) {
      // En eski kaydÄ±n zamanÄ±nÄ± al (retryAfter iÃ§in)
      const oldest = await redis.zrange(key, 0, 0, { withScores: true });
      const oldestTimestamp = oldest[0] && typeof oldest[0] === 'object' && 'score' in oldest[0] 
        ? (oldest[0] as { score: string }).score 
        : now;
      const retryAfter = Math.ceil((parseInt(String(oldestTimestamp)) + windowMs - now) / 1000);
      
      return { 
        success: false, 
        limit, 
        remaining: 0, 
        retryAfter: Math.max(1, retryAfter),
        resetTime: parseInt(String(oldestTimestamp)) + windowMs
      };
    }
    
    // 4. Yeni isteÄŸi ekle
    await redis.zadd(key, { score: now, member: `${now}:${Math.random().toString(36).slice(2)}` });
    
    // 5. Key'in Ã¶mrÃ¼nÃ¼ ayarla (window sÃ¼resi + buffer)
    await redis.expire(key, Math.ceil(windowMs / 1000) + 60);
    
    return { 
      success: true, 
      limit, 
      remaining: limit - (count + 1),
      resetTime: now + windowMs
    };
    
  } catch (error) {
    console.error('Rate limit error:', error);
    // Fail-open: hata durumunda rate limit uygulama
    return { success: true, limit, remaining: 0 };
  }
}

// ==================== GÃœVENLÄ° IP DETECTION ====================
export function getClientIp(req: NextRequest): string {
  return (
    // Vercel Forwarded-For (EN GÃœVENÄ°LÄ°R)
    req.headers.get('x-vercel-forwarded-for')?.split(',')[0]?.trim() ||
    
    // Cloudflare Real IP
    req.headers.get('cf-connecting-ip') ||
    req.headers.get('x-real-ip') ||
    
    // Standart Forwarded-For (gÃ¼venilmez, fallback)
    req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() ||
    
    // Local dev fallback
    (process.env.NODE_ENV === 'development' ? '127.0.0.1' : 'unknown')
  ) || 'unknown';
}
