import { NextRequest, NextResponse } from "next/server";
import { createUmi } from "@metaplex-foundation/umi-bundle-defaults";
import {
  genesis,
  createLaunch,
  isGenesisApiError,
  isGenesisApiNetworkError,
  isGenesisValidationError,
} from "@metaplex-foundation/genesis";

export const runtime = "nodejs";

function safeError(error: unknown) {
  if (error instanceof Error) {
    return { name: error.name, message: error.message, stack: error.stack };
  }
  return { message: String(error) };
}

// POST /api/bonding-curve/launch
// Backend sadece transaction HAZIRLAR â€” imzalamaz, SOL harcamaz
// KullanÄ±cÄ± cÃ¼zdanÄ± imzalar â†’ Genesis fee kullanÄ±cÄ±dan Ã§ekilir
// creatorFeeWallet = PLATFORM cÃ¼zdanÄ± â†’ her swap'ta fee buraya akar
export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    console.log("LAUNCH_REQUEST:", body);

    const { name, symbol, image, description, twitter, telegram, website, userWallet } = body;

    if (!name || !symbol || !image) {
      return NextResponse.json(
        { success: false, error: "Missing required fields: name, symbol, image" },
        { status: 400 }
      );
    }

    if (!userWallet) {
      return NextResponse.json(
        { success: false, error: "Missing userWallet" },
        { status: 400 }
      );
    }

    // Platform cÃ¼zdanÄ± â€” swap fee'leri buraya akar
    const platformFeeWallet = process.env.BONDING_CURVE_FEE_WALLET || process.env.PLATFORM_PUBLIC_KEY;
    if (!platformFeeWallet) {
      return NextResponse.json(
        { success: false, error: "Platform fee wallet not configured" },
        { status: 500 }
      );
    }

    const rpcUrl = process.env.NEXT_PUBLIC_RPC_URL;
    if (!rpcUrl) throw new Error("NEXT_PUBLIC_RPC_URL missing");

    const umi = createUmi(rpcUrl).use(genesis());

    const input = {
      wallet: userWallet,           // KullanÄ±cÄ± cÃ¼zdanÄ± â€” Genesis launch fee'yi O Ã¶der
      token: {
        name,
        symbol,
        image,                      // Direkt gÃ¶rsel URL (Pinata PNG/JPG)
        description: description || "",
        externalLinks: {
          website: website || undefined,
          twitter: twitter || undefined,
          telegram: telegram || undefined,
        },
      },
      launchType: "bondingCurve" as const,
      launch: {
        creatorFeeWallet: platformFeeWallet, // âœ… PLATFORM cÃ¼zdanÄ± â€” her swap'ta fee buraya gelir
        firstBuyAmount: 0,
      },
    };

    console.log("CREATE_INPUT:", {
      ...input,
      launch: { ...input.launch },
    });

    // createLaunch: sadece TX hazÄ±rlar, IMZALAMAZ
    const result = await createLaunch(umi, {}, input);

    console.log("CREATE_RESULT:", {
      mintAddress: result.mintAddress,
      genesisAccount: result.genesisAccount,
      txCount: result.transactions.length,
    });

    const serializedTxs = result.transactions.map((tx) =>
      Buffer.from(umi.transactions.serialize(tx)).toString("base64")
    );

    return NextResponse.json({
      success: true,
      transactions: serializedTxs,
      blockhash: result.blockhash,
      mintAddress: result.mintAddress,
      genesisAccount: result.genesisAccount,
    });

  } catch (err) {
    console.error("LAUNCH_FATAL:", safeError(err));

    if (isGenesisValidationError(err)) {
      return NextResponse.json(
        { success: false, type: "VALIDATION_ERROR", field: (err as any).field, error: (err as any).message },
        { status: 400 }
      );
    }
    if (isGenesisApiError(err)) {
      return NextResponse.json(
        { success: false, type: "GENESIS_API_ERROR", statusCode: (err as any).statusCode, details: (err as any).responseBody },
        { status: 502 }
      );
    }
    if (isGenesisApiNetworkError(err)) {
      return NextResponse.json(
        { success: false, type: "NETWORK_ERROR", error: (err as any).cause?.message },
        { status: 503 }
      );
    }

    return NextResponse.json(
      { success: false, type: "UNKNOWN_ERROR", error: err instanceof Error ? err.message : "Unknown error" },
      { status: 500 }
    );
  }
}
