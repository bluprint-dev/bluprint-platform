import { BlockhashWithExpiryBlockHeight, PublicKey, PublicKeyInput, RpcSendTransactionOptions, Signer, Transaction } from '@metaplex-foundation/umi';
export type SvmNetwork = 'solana-mainnet' | 'solana-devnet';
export type TimeUnit = 'SECOND' | 'MINUTE' | 'HOUR' | 'DAY' | 'WEEK' | 'TWO_WEEKS' | 'MONTH' | 'QUARTER' | 'YEAR';
/**
 * Quote mint can be a friendly name ('SOL', 'USDC').
 * The SDK resolves friendly names to the correct mint address per network.
 */
export type QuoteMintInput = 'SOL' | 'USDC';
export type CreateLaunchType = 'launchpool' | 'bondingCurve';
export interface ExternalLinks {
    /** Website URL */
    website?: string;
    /** Twitter/X handle (e.g. '@mytoken') or full URL */
    twitter?: string;
    /** Telegram handle (e.g. '@mytoken') or full URL */
    telegram?: string;
}
export interface TokenMetadata {
    /** Token name, 1-32 characters */
    name: string;
    /** Token ticker symbol, 1-10 characters */
    symbol: string;
    /** Token image URL hosted on Irys (must start with https://gateway.irys.xyz/) */
    image: string;
    /** Optional description, max 250 characters */
    description?: string;
    /** Optional social and web links */
    externalLinks?: ExternalLinks;
}
export interface LaunchpoolConfig {
    /**
     * Number of tokens to sell via the launchpool.
     * Must be a portion of the 1 billion total supply (e.g., 500_000_000).
     */
    tokenAllocation: number;
    /**
     * When the deposit period opens.
     * The deposit period lasts exactly 48 hours from this time.
     */
    depositStartTime: Date | string;
    /**
     * Minimum amount of quote tokens to raise for the launch to succeed.
     * Specified in whole units (e.g., 200 means 200 SOL, 25000 means 25000 USDC).
     */
    raiseGoal: number;
    /**
     * Percentage of raised funds allocated to Raydium LP, in basis points.
     * Minimum 2000 (20%), maximum 10000 (100%).
     * The remainder goes to the funds recipient as unlocked funds.
     */
    raydiumLiquidityBps: number;
    /**
     * Wallet address that receives the unlocked portion of raised funds.
     */
    fundsRecipient: PublicKey | string;
}
/**
 * Cliff unlock amount configuration.
 * - `'linear'` or `{ type: 'linear' }`: Uses linear vesting calculation for cliff amount
 * - `number` or `{ type: 'custom'; value: number }`: Explicit token amount released at cliff
 */
export type CliffUnlockAmount = number | 'linear' | {
    type: 'linear';
} | {
    type: 'custom';
    value: number;
};
export interface LockedAllocation {
    /** Human-readable name (e.g., 'Team', 'Advisors') for this vesting stream (max 64 chars) */
    name: string;
    /** Recipient wallet address */
    recipient: PublicKey | string;
    /** Total number of tokens in the vesting schedule */
    tokenAmount: number;
    /** When the vesting stream starts */
    vestingStartTime: Date | string;
    /** Duration of the full vesting period */
    vestingDuration: {
        value: number;
        unit: TimeUnit;
    };
    /** How frequently tokens are released */
    unlockSchedule: TimeUnit;
    /** Optional cliff configuration */
    cliff?: {
        /** Duration until the cliff */
        duration: {
            value: number;
            unit: TimeUnit;
        };
        /**
         * Tokens released at cliff.
         * - Omitted: defaults to linear vesting
         * - `number`: custom token amount (backward compatible shorthand)
         * - `'linear'`: linear vesting calculation
         * - `{ type: 'linear' }`: linear vesting calculation
         * - `{ type: 'custom', value: number }`: explicit custom token amount
         */
        unlockAmount?: CliffUnlockAmount;
    };
    /** Whether the sender can cancel the vesting stream. Defaults to false. */
    cancelableBySender?: boolean;
    /** Whether the recipient can cancel the vesting stream. Defaults to false. */
    cancelableByRecipient?: boolean;
    /** Whether the sender can transfer the vesting stream. Defaults to false. */
    transferableBySender?: boolean;
    /** Whether the recipient can transfer the vesting stream. Defaults to false. */
    transferableByRecipient?: boolean;
}
interface BaseLaunchInput {
    /** The creator's wallet public key (will sign transactions) */
    wallet: PublicKey | string;
    /** Token metadata */
    token: TokenMetadata;
    /** Network, defaults to 'solana-mainnet' */
    network?: SvmNetwork;
    /**
     * Quote token for the sale.
     * Accepts 'SOL', 'USDC', or a raw mint address string.
     * Defaults to 'SOL'.
     */
    quoteMint?: QuoteMintInput;
    /**
     * Agent configuration. When provided, the transactions will be wrapped in
     * core execute ix for the agent to execute.
     */
    agent?: {
        /** Agent NFT mint address */
        mint: PublicKeyInput;
        /** Whether to set the token on the agent */
        setToken: boolean;
    };
}
export interface CreateLaunchpoolLaunchInput extends BaseLaunchInput {
    launchType: 'launchpool';
    launch: LaunchpoolLaunchInput;
}
export interface CreateBondingCurveLaunchInput extends BaseLaunchInput {
    launchType: 'bondingCurve';
    launch: BondingCurveLaunchInput;
}
export interface PreprocessedFields {
    agentWallet?: PublicKey;
}
export type PreprocessedLaunchInput = CreateLaunchInput & PreprocessedFields;
export type CreateLaunchInput = CreateLaunchpoolLaunchInput | CreateBondingCurveLaunchInput;
/**
 * Input for creating a launchpool launch.
 */
export interface LaunchpoolLaunchInput {
    /** Launchpool (primary sale mechanism) configuration */
    launchpool: LaunchpoolConfig;
    /** Optional: locked/vesting token allocations via Streamflow */
    lockedAllocations?: LockedAllocation[];
}
/**
 * Input for creating a bonding curve launch.
 *
 * All protocol parameters (supply splits, virtual token amounts, fund flows,
 * lock schedules) are set to protocol defaults automatically.
 */
export interface BondingCurveLaunchInput {
    /**
     * Wallet address to receive creator fees.
     *
     * Defaults to the launching wallet (or agent PDA when agent mint is provided).
     */
    creatorFeeWallet?: PublicKey | string;
    /**
     * SOL amount for a mandatory first buy by the launching wallet (e.g. 0.1
     * means 0.1 SOL). When a nonzero value is provided, the first swap on the
     * bonding curve must come from the launching wallet for exactly this amount.
     *
     * Omit or set to `0` to disable the first-buy restriction.
     */
    firstBuyAmount?: number;
    /**
     * Signer for the mandatory first buy. The SDK extracts the public key for
     * the API payload and automatically co-signs transactions via
     * {@link createAndRegisterLaunch} or {@link signAndSendLaunchTransactions}.
     *
     * Defaults to the launching wallet (or agent PDA when agent mint is provided).
     */
    firstBuyWallet?: Signer;
}
export interface GenesisApiConfig {
    /** Base URL of the Metaplex API. Defaults to 'https://api.metaplex.com'. */
    baseUrl?: string;
    /** Custom fetch implementation. Defaults to globalThis.fetch. */
    fetch?: typeof globalThis.fetch;
}
export interface RegisterLaunchInput {
    /** The genesis account public key (returned by createLaunch) */
    genesisAccount: PublicKey | string;
    /** The original create launch input, used to reconstruct the payload for validation */
    createLaunchInput: CreateLaunchInput;
    /**
     * Optional launch owner override for registration (if different from
     * `createLaunchInput.wallet`). Do not use this for bonding curve creator
     * fee recipients; those are set via `launch.creatorFeeWallet` /
     * `launch.creatorFeeAgentMint` on the create payload only.
     */
    creatorWallet?: PublicKey | string;
    /** Optional Twitter verification token for verified badge */
    twitterVerificationToken?: string;
}
/** Optional fields forwarded to `registerLaunch` when using `createAndRegisterLaunch`. */
export type RegisterOptions = Omit<RegisterLaunchInput, 'genesisAccount' | 'createLaunchInput'>;
export interface CreateLaunchResponse {
    /** Deserialized Umi transactions ready to be signed and sent */
    transactions: Transaction[];
    /** Blockhash for transaction validity */
    blockhash: BlockhashWithExpiryBlockHeight;
    /** The mint address of the created token */
    mintAddress: string;
    /** The genesis account PDA address */
    genesisAccount: string;
}
export interface RegisterLaunchResponse {
    /** Whether this launch was already registered */
    existing?: boolean;
    /** Launch information */
    launch: {
        id: string;
        link: string;
    };
    /** Token information */
    token: {
        id: string;
        mintAddress: string;
    };
}
export interface ClaimCreatorRewardsInput {
    /** The wallet public key to claim creator rewards for */
    wallet: PublicKey | string;
    /**
     * Optional payer that covers rent and transaction fees for the claim.
     * Defaults to `wallet` when omitted.
     */
    payer?: PublicKey | string;
    /** Network, defaults to 'solana-mainnet' */
    network?: SvmNetwork;
}
export interface ClaimCreatorRewardsResponse {
    /** Deserialized Umi transactions ready to be signed and sent */
    transactions: Transaction[];
    /** Blockhash for transaction validity */
    blockhash: BlockhashWithExpiryBlockHeight;
}
export interface CreateAndRegisterLaunchResult {
    /** Transaction signatures from signing and sending */
    signatures: Uint8Array[];
    /** The mint address of the created token */
    mintAddress: string;
    /** The genesis account PDA address */
    genesisAccount: string;
    /** Launch information from registration */
    launch: {
        id: string;
        link: string;
    };
    /** Token information from registration */
    token: {
        id: string;
        mintAddress: string;
    };
}
export interface SignAndSendOptions extends RpcSendTransactionOptions {
    /** custom tx sender which returns the signatures of the transaction */
    txSender?: (txs: Transaction[]) => Promise<Uint8Array[]>;
    /** Additional signers to include when signing transactions (e.g. firstBuyWallet) */
    additionalSigners?: Signer[];
}
export {};
